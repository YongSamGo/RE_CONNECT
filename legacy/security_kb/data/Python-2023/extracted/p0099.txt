import secrets
import string
def generate_session_key():
  RANDOM_STRING_CHARS = string.ascii_letters+string.digits
  # 보안 기능과 관련된 난수는 secrets 라이브러리를 사용해야 안전하다
  return “”.join(secrets.choice(RANDOM_STRING_CHARS) for i in range(32))
