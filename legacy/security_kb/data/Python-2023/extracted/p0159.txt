class UserObj:
  def __init__(self):
    self.__privateVariable = []
    
  # private 배열에 외부 값을 바로 대입하는 경우 [:]를 사용하여 
  # 외부와 내부의 배열이 서로 참조되지 않도록 해야 한다
  def set_private_member(self, input_list):
    self.__privateVariable = input_list[:]
